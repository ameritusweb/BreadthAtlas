"""Blind player 1 route: a linear margin reader over the trained model's component features."""
import numpy as np
import poc4_components as C

_w = _mu = _sd = None

def _stack(feats, q):
    return np.nan_to_num(np.column_stack([feats, q]), nan=0.0, posinf=0.0, neginf=0.0)

def _fit():
    global _w, _mu, _sd
    c = C.load_cache()
    th = _stack(c["train_h"], c["train_qh"]); tl = _stack(c["train_l"], c["train_ql"])
    both = np.vstack([th, tl]); _mu = both.mean(0); _sd = both.std(0); _sd[_sd < 1e-9] = 1.0
    d = (th - tl) / _sd
    X = np.vstack([d, -d]); y = np.concatenate([np.ones(len(d)), np.zeros(len(d))])
    w = np.zeros(X.shape[1]); lam = 1.0
    for _ in range(50):  # Newton steps on L2-regularised logistic loss
        p = 1.0 / (1.0 + np.exp(-np.clip(X @ w, -30, 30)))
        g = X.T @ (p - y) + lam * w
        H = (X * (p * (1 - p))[:, None]).T @ X + lam * np.eye(X.shape[1])
        step = np.linalg.solve(H, g); w -= step
        if np.abs(step).max() < 1e-8:
            break
    _w = w

def score(question: str, answer: str) -> float:
    if _w is None:
        _fit()
    c = C.load_cache()
    try:
        row, side = C.val_row(answer)
    except KeyError:
        return 0.0
    feats = c["val_h"][row] if side == "h" else c["val_l"][row]
    q = c["val_qh"][row] if side == "h" else c["val_ql"][row]
    x = _stack(feats[None, :], np.array([q]))[0]
    return float(((x - _mu) / _sd) @ _w)
```

"""Field-evolution sign vote: each tri_evo_* component casts a train-fitted, weighted vote."""
import numpy as np
from poc4_components import load_cache, val_row, family_columns

FAMILY = "tri_evo_"
_model = None

def _fit():
    global _model
    if _model is None:
        c = load_cache()
        cols = family_columns(c["names"].tolist(), FAMILY)
        th, tl = c["train_h"][:, cols], c["train_l"][:, cols]
        win = (th > tl).mean(axis=0)
        _model = dict(cols=cols, sign=np.where(win >= 0.5, 1.0, -1.0), weight=np.abs(win - 0.5),
                      pooled=np.sort(np.vstack([th, tl]), axis=0))
    return _model

def score(question: str, answer: str) -> float:
    m = _fit(); c = load_cache()
    try:
        row, side = val_row(answer)
    except KeyError:
        return 0.0
    x = (c["val_h"] if side == "h" else c["val_l"])[row, m["cols"]]
    n = m["pooled"].shape[0]
    pct = np.array([np.searchsorted(m["pooled"][:, j], x[j]) / n for j in range(len(x))])
    return float(np.sum(m["weight"] * m["sign"] * pct))
